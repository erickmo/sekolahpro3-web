> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a banner-ads platform — managed from the SaaS admin app, served into all 6 other SekolahPro apps, tracking impressions/clicks per Property (app).

**Architecture:** The `vernon_ads` Frappe app (7 doctypes + serve/track/click API) already exists but is undeployed and has no frontend. **Tech Stack:** Frappe/Python (pytest/unittest), React 18 + TypeScript, TanStack Router/Query, `@sekolahpro/ui`, Vite, vitest + React Testing Library, pnpm workspace + Turbo.

Create `vernon_ads/api/stats.py`:
```python
"""Aggregated ad metrics for the SaaS admin dashboard.

Read-only GROUP BY aggregation over Ad Event. Exposed as JSON for the React
admin (a native Frappe Query Report cannot return cleanly to the SPA), so this
lives as a thin whitelisted method delegating to module helpers — Frappe
low-code Priority 5. No doctype lifecycle hook fits a cross-doctype read.
"""
import frappe
from frappe.utils import add_days, nowdate

DEFAULT_RANGE_DAYS = 30


@frappe.whitelist()
def overview(from_date: str = "", to_date: str = "") -> dict:
    """Return totals + per-campaign + per-property + daily series in [from_date, to_date].
    Defaults to the last DEFAULT_RANGE_DAYS days. Requires an authenticated session."""
    to_date = to_date or nowdate()
    from_date = from_date or add_days(to_date, -DEFAULT_RANGE_DAYS)
    return {
        "from_date": from_date,
        "to_date": to_date,
        "totals": _totals(from_date, to_date),
        "by_campaign": _by_campaign(from_date, to_date),
        "by_property": _by_property(from_date, to_date),
        "daily": _daily(from_date, to_date),
    }


# Inclusive date window on the read-only `timestamp`. `< to_date + 1 day` keeps
# the whole end day. %(params)s are bound by frappe.db.sql (no SQL injection).
_WINDOW = "e.timestamp >= %(from_date)s AND e.timestamp < DATE_ADD(%(to_date)s, INTERVAL 1 DAY)"
_IMP = "SUM(e.event_type = 'Impression')"
_CLK = "SUM(e.event_type = 'Click')"


def _totals(from_date: str, to_date: str) -> dict:
    row = frappe.db.sql(
        f"SELECT {_IMP} AS impressions, {_CLK} AS clicks "
        f"FROM `tabAd Event` e WHERE {_WINDOW}",
        {"from_date": from_date, "to_date": to_date},
        as_dict=True,
    )[0]
    return {"impressions": int(row.impressions or 0), "clicks": int(row.clicks or 0)}


def _by_campaign(from_date: str, to_date: str) -> list:
    return frappe.db.sql(
        f"SELECT c.campaign AS campaign, {_IMP} AS impressions, {_CLK} AS clicks "
        f"FROM `tabAd Event` e JOIN `tabAd Creative` c ON c.name = e.ad_creative "
        f"WHERE {_WINDOW} GROUP BY c.campaign ORDER BY impressions DESC",
        {"from_date": from_date, "to_date": to_date},
        as_dict=True,
    )


def _by_property(from_date: str, to_date: str) -> list:
    return frappe.db.sql(
        f"SELECT s.property AS property, {_IMP} AS impressions, {_CLK} AS clicks "
        f"FROM `tabAd Event` e JOIN `tabAd Slot` s ON s.name = e.ad_slot "
        f"WHERE {_WINDOW} GROUP BY s.property ORDER BY impressions DESC",
        {"from_date": from_date, "to_date": to_date},
        as_dict=True,
    )


def _daily(from_date: str, to_date: str) -> list:
    return frappe.db.sql(
        f"SELECT DATE(e.timestamp) AS day, {_IMP} AS impressions, {_CLK} AS clicks "
        f"FROM `tabAd Event` e WHERE {_WINDOW} GROUP BY DATE(e.timestamp) ORDER BY day",
        {"from_date": from_date, "to_date": to_date},
        as_dict=True,
    )
```

Create `vernon_ads/setup/install.py`:
```python
"""Idempotent seed of the SekolahPro consuming apps as ad Properties.

Runs from the `after_install` framework hook (Frappe low-code Priority 3).
NOT shipped as fixtures: Property.api_key is a per-install secret
(secrets.token_hex, read_only, no_copy) that must never be committed.
"""
import frappe

PROPERTY_GROUP = "SekolahPro Apps"

# property_name -> public site URL (informational; ad serving keys off api_key)
APPS = {
    "school": "http://localhost:5181",
    "parent": "http://localhost:5184",
    "student": "http://localhost:5182",
    "merchant": "http://localhost:5184",
    "landing": "http://localhost:5173",
    "situs": "http://localhost:5184",
}

# Starter banner slots created per property. (slot_key suffix, width, height)
STARTER_SLOTS = [
    ("dashboard-top", 728, 90),
    ("sidebar", 300, 250),
    ("content-bottom", 728, 90),
]
AD_TYPE_BANNER = "Banner"


def after_install():
    """Frappe entry point. Wrapped so a seed failure never aborts app install."""
    try:
        seed_apps()
    except Exception:
        frappe.log_error(title="vernon_ads seed failed")


def seed_apps():
    """Create the property group, one Property per consuming app, and starter
    banner slots. Idempotent — safe to re-run; existing docs are left untouched."""
    if not frappe.db.exists("Property Group", PROPERTY_GROUP):
        frappe.get_doc({
            "doctype": "Property Group",
            "name": PROPERTY_GROUP,
            "description": "Aplikasi SekolahPro yang menayangkan banner",
        }).insert(ignore_permissions=True)

    for name, url in APPS.items():
        _ensure_property(name, url)
        for suffix, width, height in STARTER_SLOTS:
            _ensure_slot(name, suffix, width, height)


def _ensure_property(name: str, url: str):
    if frappe.db.exists("Property", name):
        return
    frappe.get_doc({
        "doctype": "Property",
        "property_name": name,
        "platform": "Web",
        "property_group": PROPERTY_GROUP,
        "status": "Active",
        "url": url,
    }).insert(ignore_permissions=True)


def _ensure_slot(property_name: str, suffix: str, width: int, height: int):
    slot_key = f"{property_name}:{suffix}"
    if frappe.db.exists("Ad Slot", slot_key):
        return
    frappe.get_doc({
        "doctype": "Ad Slot",
        "slot_key": slot_key,
        "property": property_name,
        "ad_type": AD_TYPE_BANNER,
        "width": width,
        "height": height,
    }).insert(ignore_permissions=True)
```

import type { AdCreative } from "./types";

const GET_AD = "/api/method/vernon_ads.api.get_ad.get_ad";
const TRACK = "/api/method/vernon_ads.api.track.track";
// Guest CSRF sentinel mirrors the backend serve.js so guest POSTs pass Frappe's
// CSRF check without a session cookie.
const GUEST_CSRF = "Guest";

/** Fetch a creative for a slot. Returns null on no-ad / error (caller renders nothing). */
export async function fetchAd(
  baseUrl: string,
  slot: string,
  propertyKey: string,
): Promise<AdCreative | null> {
  const url =
    `${baseUrl}${GET_AD}?slot=${encodeURIComponent(slot)}` +
    `&property_key=${encodeURIComponent(propertyKey)}`;
  try {
    const res = await fetch(url, { method: "GET" });
    if (!res.ok) return null;
    const json = (await res.json()) as { message?: AdCreative | null };
    return json.message ?? null;
  } catch {
    return null;
  }
}

/** Record an impression. Fire-and-forget; failures are swallowed. */
export async function trackImpression(baseUrl: string, token: string): Promise<void> {
  try {
    await fetch(`${baseUrl}${TRACK}`, {
      method: "POST",
      headers: { "Content-Type": "application/json", "X-Frappe-CSRF-Token": GUEST_CSRF },
      body: JSON.stringify({ token }),
    });
  } catch {
    /* analytics best-effort */
  }
}

/** Resolve a click into its destination URL via the backend (records the click). */
export async function resolveClick(baseUrl: string, clickUrl: string): Promise<string> {
  try {
    const res = await fetch(`${baseUrl}${clickUrl}`, { method: "GET" });
    const json = (await res.json()) as { message?: { redirect?: string } };
    return json.message?.redirect ?? "";
  } catch {
    return "";
  }
}

import { useEffect, useState } from "react";
import { fetchAd } from "./client";
import { useAdsConfig } from "./AdsProvider";
import type { AdCreative } from "./types";

export interface UseAdResult {
  creative: AdCreative | null;
  loading: boolean;
}

/** Fetch the creative for `slot` once on mount. No-op (null) when config or
 * propertyKey is missing, so apps render safely before ads are configured. */
export function useAd(slot: string): UseAdResult {
  const cfg = useAdsConfig();
  const [creative, setCreative] = useState<AdCreative | null>(null);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    if (!cfg || !cfg.propertyKey) return;
    let active = true;
    setLoading(true);
    void fetchAd(cfg.baseUrl, slot, cfg.propertyKey).then((ad) => {
      if (!active) return;
      setCreative(ad);
      setLoading(false);
    });
    return () => {
      active = false;
    };
  }, [cfg, slot]);

  return { creative, loading };
}

import { useEffect, useRef } from "react";
import { cn } from "@sekolahpro/ui";
import { useAdsConfig } from "./AdsProvider";
import { useAd } from "./useAd";
import { trackImpression, resolveClick } from "./client";

const VISIBLE_RATIO = 0.5;

interface Props {
  /** Slot key as seeded, e.g. "school:dashboard-top". */
  slot: string;
  className?: string;
}

/** Image banner for a slot. Fetches a creative, fires one impression when ≥50%
 * visible, and routes clicks through the backend (which records + redirects).
 * Renders nothing when there is no ad. */
export function AdBanner({ slot, className }: Props) {
  const cfg = useAdsConfig();
  const { creative } = useAd(slot);
  const ref = useRef<HTMLAnchorElement>(null);
  const tracked = useRef(false);

  useEffect(() => {
    if (!creative || !cfg || tracked.current) return;
    const el = ref.current;
    if (!el) return;
    const obs = new IntersectionObserver(
      (entries) => {
        if (tracked.current) return;
        if (entries.some((e) => e.isIntersecting)) {
          tracked.current = true;
          void trackImpression(cfg.baseUrl, creative.token);
          obs.disconnect();
        }
      },
      { threshold: VISIBLE_RATIO },
    );
    obs.observe(el);
    return () => obs.disconnect();
  }, [creative, cfg]);

  if (!cfg || !creative || !creative.image_url) return null;

  async function handleClick(e: React.MouseEvent) {
    e.preventDefault();
    const dest = await resolveClick(cfg!.baseUrl, creative!.click_url);
    if (dest) window.open(dest, "_blank", "noopener,noreferrer");
  }

  return (
    <a
      ref={ref}
      href={creative.click_url}
      onClick={handleClick}
      className={cn("inline-block max-w-full", className)}
      data-ad-slot={slot}
    >
      <img
        src={creative.image_url}
        alt={creative.title ?? "Iklan"}
        width={creative.width ?? undefined}
        height={creative.height ?? undefined}
        className="block max-w-full h-auto"
      />
    </a>
  );
}

import type { CrudConfig, CrudRow } from "./types";

/** Strip empty strings so we don't overwrite Frappe fields with "". */
function clean(values: Record<string, string>): Record<string, unknown> {
  const out: Record<string, unknown> = {};
  for (const [k, v] of Object.entries(values)) if (v !== "") out[k] = v;
  return out;
}

export function ResourceCrudPage({ config }: { config: CrudConfig }) {
  const qc = useQueryClient();
  const listQ = useResourceList<CrudRow>(config.doctype, {
    fields: config.listFields, limit_page_length: 200, order_by: "modified desc",
  });
  const createM = useResourceCreate(config.doctype);
  const updateM = useResourceUpdate(config.doctype);
  const deleteM = useResourceDelete(config.doctype);
  const [editing, setEditing] = useState<CrudRow | null | undefined>(undefined); // undefined=closed, null=create

  const invalidate = () => qc.invalidateQueries({ queryKey: ["resource:list", config.doctype] });

  function submit(values: Record<string, string>) {
    const body = clean(values);

    if (editing) updateM.mutate({ name: editing.name, patch: body }, done);
    else createM.mutate(body, done);

        </div>

      <PageHeader
        title={config.title}
        description={config.description}

          </Button>
        }
      />

        <DataTable<CrudRow>
          data={listQ.data ?? []}
          columns={columns}

/** Simple CSS bar series (no chart lib in @sekolahpro/ui). Height ∝ impressions. */
function MiniBars({ daily }: { daily: AdsDaily[] }) {
  const max = Math.max(1, ...daily.map((d) => d.impressions));
  if (!daily.length) return <div className="text-sm text-muted-fg">Belum ada data.</div>;
  return (
    <div className="flex items-end gap-1 h-32">
      {daily.map((d) => (
        <div key={d.day} className="flex-1 bg-brand/70 rounded-t" style={{ height: `${(d.impressions / max) * 100}%` }} title={`${d.day}: ${d.impressions} impr / ${d.clicks} klik`} />
      ))}
    </div>
  );
}

function AdsDashboard() {
  const [data, setData] = useState<AdsOverview | null>(null);
  const [error, setError] = useState(false);

  useEffect(() => {
    let active = true;
